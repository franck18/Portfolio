# Exemple — création de compte AD depuis CSV
Import-Csv "utilisateurs.csv" | ForEach-Object {
  New-ADUser -Name $_.Nom -SamAccountName $_.Login
    -Path "OU=Stagiaires,DC=mairie,DC=local"
    -AccountPassword (ConvertTo-SecureString $_.MotDePasse -AsPlainText -Force)
    -Enabled $true
}

# Exemple — alerte si disque > 80%
USAGE=$(df / | awk 'NR==2 {print $5}' | tr -d '%')
if [ $USAGE -gt 80 ]; then
  echo "ALERTE disque : ${USAGE}% utilisé" | mail -s "Disque critique" admin@local
fi
